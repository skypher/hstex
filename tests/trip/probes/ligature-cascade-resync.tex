% Where the rebuilt run stops, in the trip font.
%
% `a' is inert in this font: it starts no ligature program and nothing
% ligatures into it, so it interrupts the cascade.  The word is 1512aaa and
% the break falls after 15.  If the replace count ran to the end of the word
% it would take in the three a's; it does not.
%
% Run from a directory holding trip.tfm (tests/trip/run-trip.sh builds one in
% build/trip):
%
%   pdftex -ini -interaction=nonstopmode ligature-cascade-resync.tex
%
% See docs/DECISIONS.md, a-break-inside-a-cascade-of-ligatures.
%
\catcode`\{=1 \catcode`\}=2 \catcode`\#=6
\font\rip=trip \rip
\lccode`A=`1 \lccode`B=`5 \lccode`C=`2
\lccode`1=`1 \lccode`5=`5 \lccode`2=`2 \lccode`7=`7
\lccode`a=`a
\language=0
\patterns{AB1AC}
\lefthyphenmin=1 \righthyphenmin=1
\hyphenchar\rip=`-
\showboxbreadth=9999 \showboxdepth=9999 \tracingonline=1
\hbadness=10000 \hfuzz=1000pt
\hyphenpenalty=10000 \exhyphenpenalty=10000
\setbox0=\vbox{\hsize=400pt \parindent=0pt \parfillskip=0pt
\rightskip=0pt plus 1fil \pretolerance=-1 \tolerance=10000
7 1512aaa\par}
\showbox0
\end
