% WHEN BRACES AROUND ONE ATOM FALL AWAY.
%
% `{x}' in a formula is the atom x: the braces make no sub-formula of their
% own, so the atom carries scripts exactly as x does. A list that turned
% into a FRACTION is not one atom, however few noads stand on either side
% of the \over -- `{B\over}' is a fraction whose denominator is empty, and
% `{\over C}' one whose numerator is:
%
%   $A^{B\over}$      ^\fraction, thickness = default
%                     ^\\mathord
%                     ^\.\fam1 B
%                     ^/{}
%
% This engine let the braces fall away and lost the fraction with them.
% trip line 283 writes {\above9pt{v\overwithdelims..
\catcode`\{=1 \catcode`\}=2 \catcode`\$=3 \catcode`\^=7 \catcode`\_=8
\tracingonline=1 \showboxdepth=99 \showboxbreadth=99
\font\r=cmr10 \font\s=cmsy10 \font\x=cmex10 \r
\textfont0=\r \scriptfont0=\r \scriptscriptfont0=\r
\textfont1=\r \scriptfont1=\r \scriptscriptfont1=\r
\textfont2=\s \scriptfont2=\s \scriptscriptfont2=\s
\textfont3=\x \scriptfont3=\x \scriptscriptfont3=\x
\message{[A a fraction inside a superscript]}
\setbox0=\hbox{$A^{B\over}\showlists$}
\message{[B one inside a braced group]}
\setbox0=\hbox{$A{B\over C}\showlists$}
\message{[C one inside a subscript, empty numerator]}
\setbox0=\hbox{$A_{\over C}\showlists$}
\message{[done]}
\end
