% Where the run that sets a word again begins: a word that did NOT meet the
% boundary beyond its left end is set again with its first two letters in one
% run, so a break between them is not one that can be taken -- unless those
% two letters are the whole word, where the run reaches the word's end and
% the break stands after all.

\language99 \patterns{0.111}
\language98
\message{[A a word that met the boundary]}
\setbox1=\vbox{\noindent\ 1111 \par}
\message{[B one that did not]}
\setbox1=\vbox{\noindent\ \noboundary1111 \par}
\message{[C one of five letters]}
\setbox1=\vbox{\noindent\ \noboundary11111 \par}
\message{[D one of two, whose only break stands]}
\setbox1=\vbox{\noindent\ \noboundary11 \par}
\message{[E a break after the first letter alone, with the boundary]}
\language99
\setbox1=\vbox{\noindent\ 111 \par}
\message{[F and without it, where it is lost]}
\language99 \setbox1=\vbox{\noindent\ \noboundary111 \par}
\message{[done]}
