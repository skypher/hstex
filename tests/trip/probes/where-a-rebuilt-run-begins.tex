%
%   pdftex -ini -interaction=nonstopmode where-a-rebuilt-run-begins
%

%

\lccode`\f=`\f \lccode`\i=`\i \lccode`\l=`\l \lccode`\m=`\m \lccode`\n=`\n
\lccode`\o=`\o \lccode`\r=`\r \lccode`\s=`\s \lccode`\t=`\t \lccode`\v=`\v
