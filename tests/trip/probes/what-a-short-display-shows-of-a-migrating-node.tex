% WHAT A SHORT DISPLAY SHOWS OF A MIGRATING NODE.
%
% An insertion, a mark and a \vadjust stand in a HORIZONTAL list until the
% line they are on is packed, and only then move out of it. A short
% display -- what \tracingparagraphs writes of the text between two
% breakpoints, and what a box that did not fit is headed by -- shows each of
% them the way it shows a box, as `[]':
%
%   \setbox0=\vbox{A\insert100{\hbox{}}\mark{m}A\par}
%   []\rip A[][]A
%
%   \setbox0=\vbox{A\vadjust{\kern3pt}A\par}
%   []\rip A[]A
%
% HSTeX showed a box, an unset box and a whatsit that way and passed over
% these three in silence, so the text it wrote of a paragraph was short of
% whatever had migrated out of it. trip's \everypar puts an \insert and a
% \mark in every paragraph it makes.
\catcode`\{=1 \catcode`\}=2
\tracingonline=1 \tracingparagraphs=1 \showboxbreadth=10 \showboxdepth=10
\font\rip=trip \rip
\hsize=100pt \parindent=0pt \hbadness=10000
\count100=1000 \dimen100=50pt \skip100=3pt
\message{[A an insert and a mark inside a paragraph]}
\setbox0=\vbox{A\insert100{\hbox{}}\mark{m}A\par}
\showbox0
\message{[B a vadjust]}
\setbox0=\vbox{A\vadjust{\kern3pt}A\par}
\showbox0
\message{[C a whatsit and a box, which were shown already]}
\setbox0=\vbox{A\special{x}\hbox{}A\par}
\showbox0
\message{[done]}
\end
