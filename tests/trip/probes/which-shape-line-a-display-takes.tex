% far, plus one -- and that count carries on past an EARLIER display in the
% same paragraph rather than starting again from the lines just broken.

\message{[A the first display of a paragraph]}
\setbox0=\vbox{\shape\noindent A$$\message{[width \the\displaywidth, indent \the\displayindent]}$$
\message{[B a second one, after the first has counted three]}
A$$\message{[width \the\displaywidth, indent \the\displayindent]}$$
\message{[C and a third]}
A$$\message{[width \the\displaywidth, indent \the\displayindent]}$$ A\par}
\message{[D a display that starts a paragraph]}
\setbox0=\vbox{\shape\noindent$$\message{[width \the\displaywidth, indent \the\displayindent]}$$\par}
\message{[done]}
