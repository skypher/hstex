% AN INSERTION'S CORRECTION GLUE GOES ON THE PAGE, NOT ONLY AGAINST THE GOAL.
%
% When the FIRST insertion of a class arrives on a page, \skip<n> is the
% correction glue for that class, and three things happen to it at once:
%
%   the goal loses its WIDTH        (along with room for the insertion)
%   the page gains its STRETCH      at that stretch's own order
%   the page gains its SHRINK
%
% Measured with \skip100=5pt plus 3filll minus 9pt and \dimen100=50pt
% \count100=1000, an insertion four points tall:
%
%   % t=0.0 g=100.0 b=10000 p=1 c=100000#     before it
%   % t=0.0 plus 3.0filll minus 9.0 g=91.0 b=0 p=2 c=2#   after it
%
% 100 - 5 (the width) - 4 (the insertion) = 91, and the stretch and the
% shrink are now the PAGE's. hstex had the goal right and the page
% totals bare, so every break after an insertion was judged as though
% the page could not stretch at all: b=10000 where the reference has
% b=0. That is a page-breaking difference, not a reporting one.
%
% ONLY THE FIRST OF A CLASS. A second insertion of the same class takes
% room from the goal and adds no more correction glue.
%
% AN INSERTION THAT MUST BE SPLIT SAYS SO, AND THE BREAK COSTS THE PAGE.
%
%   % split100 to 20.0,0.0 p=-101
%
% -- the class, how much room there was for it, how much of it went, and
% what the break was worth. That last is the penalty standing at the
% break and nothing else: a break at glue reports p=0, one at \penalty55
% reports p=55. It is then ADDED TO \insertpenalties, where it enters
% the cost of every break on the page after it:
%
%   % t=0.0 plus 3.0filll minus 9.0 g=1470.63013 b=0 p=0 c=-101#
%
% is trip's own, and the -101 in the cost is the -101 of the split.
% hstex kept \insertpenalties at nought, so every one of those costs was
% out by the same amount.
%
% HOW MUCH ROOM A CLASS HAS is the room left on the page DIVIDED BY THE
% FACTOR AND THEN TAKEN A THOUSAND TIMES, which is not the same number
% as multiplying first. A hundred points of room at \count200=333:
%
%   % split200 to 300.29297,0.0 p=0      divide first
%          not  300.3003                 multiply first
%
% -- 6553600/333 = 19680 (the remainder dropped), and 19680*1000 sp is
% 300.29297pt. Measured at 333, 1500, 1, 999 and 1000, and at a negative
% \vsize, which is trip's own `% split200 to -51.11694,0.0'. A factor of
% 1000 is not scaled at all, and one of nought or less means the page
% puts no limit on the class.
%
% AN INFINITE SHRINK COUNTS AS ITS AMOUNT. \skip101=1pt minus 2fil draws
%
%   ! Infinite glue shrinkage inserted from \skip101.
%
% and the page gains 2.0 of finite shrink -- the register itself is left
% alone.
\tracingonline=1 \tracingpages=1
\output={\shipout\box255 }
\vsize=100pt \maxdepth=2pt \topskip=0pt
\dimen100=50pt \count100=1000 \skip100=5pt plus 3filll minus 9pt
\dimen101=50pt \count101=500 \skip101=1pt minus 2fil
\message{[A first insert of a class]}
\hbox{}\penalty1
\insert100{\hbox{}\vskip4pt}
\penalty2
\message{[B second of the same class]}
\insert100{\hbox{}\vskip4pt}
\penalty3
\message{[C a class whose skip shrinks infinitely]}
\insert101{\hbox{}\vskip4pt}
\penalty4
\penalty-10000
\message{[D what \skip101 still reads]}\showthe\skip101
\message{[E an insertion that must split]}
\splittopskip=0pt
\dimen102=20pt \count102=1000 \skip102=0pt
\dimen103=20pt \count103=1000 \skip103=0pt
\dimen104=20pt \count104=1000 \skip104=0pt
\hbox{}\penalty20
\insert102{\hbox to 3pt{}\penalty-101 \vskip10pt\hbox to 3pt{}\penalty-7
  \vskip10pt\hbox to 3pt{}}
\penalty21
\message{[F insertpenalties]}\showthe\insertpenalties
\insert103{\hbox to 3pt{}\vskip10pt\hbox to 3pt{}\vskip10pt\hbox to 3pt{}%
  \vskip10pt\hbox to 3pt{}}
\penalty22
\insert104{\hbox to 3pt{}\penalty55 \vskip10pt\hbox to 3pt{}\penalty66
  \vskip10pt\hbox to 3pt{}}
\penalty23
\message{[G insertpenalties]}\showthe\insertpenalties
\penalty-10000
\message{[H how much room a class has]}
\dimen200=10000pt \count200=333 \skip200=0pt
\dimen201=10000pt \count201=1500 \skip201=0pt
\dimen202=10000pt \count202=1 \skip202=0pt
\dimen203=10000pt \count203=1000 \skip203=0pt
\dimen204=10000pt \count204=999 \skip204=0pt
\vsize=100pt
\hbox{}\insert200{\hbox to 3pt{}\vskip1000pt\hbox to 3pt{}}\penalty-10000
\hbox{}\insert201{\hbox to 3pt{}\vskip1000pt\hbox to 3pt{}}\penalty-10000
\hbox{}\insert202{\hbox to 3pt{}\vskip1000pt\hbox to 3pt{}}\penalty-10000
\hbox{}\insert203{\hbox to 3pt{}\vskip1000pt\hbox to 3pt{}}\penalty-10000
\hbox{}\insert204{\hbox to 3pt{}\vskip1000pt\hbox to 3pt{}}\penalty-10000
\vsize=-33pt
\hbox{}\insert200{\hbox to 3pt{}\vskip1000pt\hbox to 3pt{}}\penalty-10000
\message{[done]}
\tracingpages=0
\end
