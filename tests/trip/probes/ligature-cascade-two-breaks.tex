% Two hyphen breaks inside a cascade of ligatures, in the trip font.
%
% \hyphenpenalty=10000 forbids the breaks as breakpoints without stopping
% hyphenation from inserting them, so \showbox shows each discretionary whole
% -- its replace count and both of its lists -- instead of an emptied one.
%
% Run from a directory holding trip.tfm (tests/trip/run-trip.sh builds one in
% build/trip):
%
%   pdftex -ini -interaction=nonstopmode ligature-cascade-two-breaks.tex
%
% See docs/DECISIONS.md, a-break-inside-a-cascade-of-ligatures.
%
\catcode`\{=1 \catcode`\}=2 \catcode`\#=6
\font\rip=trip \rip
% Pattern letters stand for digits: A=1, B=5, C=2.  Digits map to themselves
% so the word's characters and the patterns' letters meet on the same codes.
\lccode`A=`1 \lccode`B=`5 \lccode`C=`2
\lccode`1=`1 \lccode`5=`5 \lccode`2=`2 \lccode`7=`7
\language=0
\patterns{AB1A}
\hyphenpenalty=10000 \exhyphenpenalty=10000 \lefthyphenmin=1 \righthyphenmin=1
\hyphenchar\rip=`-
\showboxbreadth=9999 \showboxdepth=9999 \tracingonline=1
\hbadness=10000 \hfuzz=1000pt
\setbox0=\vbox{\hsize=400pt \parindent=0pt \parfillskip=0pt
\rightskip=0pt plus 1fil \pretolerance=-1 \tolerance=10000
7 151512\par}
\showbox0
\end
