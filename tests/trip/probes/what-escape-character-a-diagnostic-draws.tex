% WHAT ESCAPE CHARACTER A DIAGNOSTIC DRAWS.
%
% \escapechar stands wherever a NAME is written, and a box a diagnostic
% draws is written the same way a fault's words are. With \escapechar`|
% the reference writes
%
%   |hbox(6.83331+1.0)x10.0
%   .|r A
%   .|glue(|baselineskip) 0.0
%   .|discretionary replacing 1
%   |mathon  |penalty 3  |kern 2.0  |mark{m}  |insert100, natural size ...
%
% and the same in \showbox, \showlists, the box a page builder deletes, and
% the short display a paragraph trace writes.
%
% What is NOT drawn with it is the WORDS around those boxes, which carry a
% backslash of their own:
%
%   Overfull \hbox (13.22226pt too wide) detected at line ...
%   > \box0=
%   \openout1 = `...'.
%   \font\nullfont=nullfont          (what a format dump writes)
%
% This engine drew a backslash everywhere. trip line 432 sets
% \escapechar`| and then ships a box out with \tracingoutput.
\catcode`\{=1 \catcode`\}=2 \catcode`\$=3 \catcode`\^=7 \catcode`\_=8
\tracingonline=1 \showboxdepth=9 \showboxbreadth=9
\font\r=cmr10 \font\s=cmsy10 \font\x=cmex10 \r
\textfont0=\r \scriptfont0=\r \scriptscriptfont0=\r
\textfont1=\r \scriptfont1=\r \scriptscriptfont1=\r
\textfont2=\s \scriptfont2=\s \scriptscriptfont2=\s
\textfont3=\x \scriptfont3=\x \scriptscriptfont3=\x
\escapechar=`|
\message{[A a box shown with a changed escape character]}
\setbox1=\hbox{A}
\setbox0=\vbox{\hrule\hbox to 10pt{A\hskip 1pt plus 2pt\lower1pt\copy1\discretionary{A}{B}{C}}
  \penalty 3\kern2pt\mark{m}\vadjust{}\insert100{}\lineskip=1pt \hbox{}\hbox{}}
\showbox0
\message{[B a formula]}
\setbox0=\hbox{$A\over B\mathchoice{A}{B}{C}{D}\radical"270370 A\underline{A}\left(A\right)$}
\showbox0
\message{[C the whatsits]}
\setbox2=\hbox{\write3{a\relax b}\openout4=zzesc \closeout4 \special{s}}
\showbox2
\message{[D a paragraph trace and the page]}
\tracingparagraphs=1 \hsize=10pt \parindent=0pt \hbadness=10000
\vsize=20pt \tolerance=10000
AAA A\par
\showlists
\message{[E and one with the escape character back]}
\escapechar=`\\
\showbox1
\message{[done]}
\end
