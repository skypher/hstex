% \tracingrestores.
%
% It was what was left of trip's log after \tracingcommands: 257 lines of
% the oracle's 7292, against none here. This engine now writes 251 of
% them and the whole log has gone from 5833 lines to 6116.
%
% One line per save undone when a group ends, LIFO:
%
%   {restoring \a=macro:->A}            a meaning, written as \meaning does
%   {restoring \two=undefined}          a meaning that was not there before
%   {restoring \gg=\mathchar"321D}      any other meaning, the same way
%   {restoring \catcode91=12}           a code, by number
%   {restoring \lccode65=66}            the other code tables likewise
%   {restoring \count5=7}               a register, by number
%   {restoring \dimen6=1.0pt}
%   {restoring \skip7=2.0pt plus 1.0fil}
%   {restoring \muskip8=3.0mu}
%   {restoring \toks9=x}                the token list's text
%   {restoring \hbadness=3}             an integer parameter, BY NAME
%   {restoring \parindent=4.0pt}        a dimen parameter
%   {restoring \baselineskip=10.0pt}    a glue parameter
%   {restoring \thinmuskip=5.0mu}       a muglue parameter
%   {restoring \everypar=p}             a token parameter
%   {restoring \textfont2=\gg}          a math family font, naming the font
%   {restoring \parshape=1}             just the count of pairs
%
% A BOX TAKES TWO LINES, the closing brace going after the box:
%
%   {restoring \box10=
%   \hbox(6.83331+0.0)x7.50002 []}
%
% and the box is written SHALLOW -- `[]' for its contents -- whatever
% \showboxbreadth and \showboxdepth say.
%
% TWO SAVE KINDS DRAW NOTHING: what \aftergroup put by, and the current
% font. The reference traces neither.
%
% TWO TRAPS worth knowing:
%   * glue_parameter_name() is written for the field a glue NODE carries,
%     where nought means `not from a parameter', so it counts from one.
%     Passing a save's index straight to it names the parameter before the
%     one that was saved -- \parfillskip for \baselineskip.
%   * A family's font is named by the control sequence \font gave it,
%     which is what \the\textfont reports, and not by its metrics file:
%     {restoring \textfont2=\gg}, not =\cmr10.
%
% NOT YET: this engine saves a value again on a second local assignment in
% the same group, where the reference saves only what was there before the
% group began. `\font\gg=cmr10' sets the control sequence twice on
% purpose -- once to \nullfont so that the `at' clause is read with the
% target already a font rather than a macro -- and so draws two restore
% lines here and one in the reference. Skipping the save where the value's
% level is already the current group level is the fix, and it wants doing
% in each of the assign_ functions.
\tracingonline=1

\muskip8=3mu \toks9={x}\hbadness=3 \parindent=4pt
\baselineskip=10pt \thinmuskip=5mu \everypar={p}\textfont2=\ff
\tracingrestores=1
{\def\a{B}\catcode`\[=1 \count5=9 \dimen6=2pt \skip7=3pt \muskip8=4mu
 \toks9={y}\hbadness=4 \parindent=5pt \baselineskip=11pt \thinmuskip=6mu
 \everypar={q}\textfont2=\nullfont \def\new{N}}
\end
