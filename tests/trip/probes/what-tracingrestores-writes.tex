% \tracingrestores.
%
% It was what was left of trip's log after \tracingcommands: 257 lines of
% the oracle's 7292, against none here. This engine now writes 251 of
% them and the whole log has gone from 5833 lines to 6116.
%
% One line per save undone when a group ends, LIFO:
%
%   {restoring \a=macro:->A}            a meaning, written as \meaning does
%   {restoring \two=undefined}          a meaning that was not there before
%   {restoring \gg=\mathchar"321D}      any other meaning, the same way
%   {restoring \catcode91=12}           a code, by number
%   {restoring \lccode65=66}            the other code tables likewise
%   {restoring \count5=7}               a register, by number
%   {restoring \dimen6=1.0pt}
%   {restoring \skip7=2.0pt plus 1.0fil}
%   {restoring \muskip8=3.0mu}
%   {restoring \toks9=x}                the token list's text
%   {restoring \hbadness=3}             an integer parameter, BY NAME
%   {restoring \parindent=4.0pt}        a dimen parameter
%   {restoring \baselineskip=10.0pt}    a glue parameter
%   {restoring \thinmuskip=5.0mu}       a muglue parameter
%   {restoring \everypar=p}             a token parameter
%   {restoring \textfont2=\gg}          a math family font, naming the font
%   {restoring \parshape=1}             just the count of pairs
%
% A BOX TAKES TWO LINES, the closing brace going after the box:
%
%   {restoring \box10=
%   \hbox(6.83331+0.0)x7.50002 []}
%
% and the box is written SHALLOW -- `[]' for its contents -- whatever
% \showboxbreadth and \showboxdepth say.
%
% TWO SAVE KINDS DRAW NOTHING: what \aftergroup put by, and the current
% font. The reference traces neither.
%
% TWO TRAPS worth knowing:
%   * glue_parameter_name() is written for the field a glue NODE carries,
%     where nought means `not from a parameter', so it counts from one.
%     Passing a save's index straight to it names the parameter before the
%     one that was saved -- \parfillskip for \baselineskip.
%   * A family's font is named by the control sequence \font gave it,
%     which is what \the\textfont reports, and not by its metrics file:
%     {restoring \textfont2=\gg}, not =\cmr10.
%
% A SECOND LOCAL ASSIGNMENT IN THE SAME GROUP SAVES NOTHING: what is kept
% is what stood before the group began. `\font\gg=cmr10' sets the control
% sequence twice on purpose -- once to \nullfont so that the `at' clause
% is read with the target already a font rather than a macro -- and draws
% one restore line, not two.
%
% THE FONT IN USE DRAWS A LINE TOO: `{restoring current font=\ff}',
% naming the control sequence \font gave it. That went unwritten for six
% turns because every probe for it was run under the PRELOADED pdftex
% format, where selecting a font is an assignment like any other and the
% eTeX rule about assigning what is already there hides it. Under
% `pdftex -ini', which is what trip runs, it is drawn every time --
% including where the same font is selected again, and where the group
% is a box. See docs/DECISIONS.md, which-tex-hstex-is-measured-against. Six shapes measured, including two
% assignments to the same register, a nested group, and a \global one
% before a local one.
%
% A RESTORE NEVER CARRIES THE MODE, however long it is since one was
% named: not one of the oracle's 257 lines has a `vertical mode: ' on it,
% where a \tracingcommands line would.
%
% A RESTORE IS SHOWN TO THIRTY-TWO BYTES AND NO FURTHER, and then says
% `\ETC.'. What is counted is the value alone -- for a macro, the
% parameter text, the `->' between and the replacement, not the `macro:'
% in front of them nor the \long and \outer before that:
%
%   \toks2 of 32 characters      shown whole
%   \toks3 of 33 characters      32 of them and `\ETC.'
%   macro:#1->0123...            #1(2) + ->(2) + 28 = 32, then `\ETC.'
%   \long\outer macro:32#1->...  32#1(4) + ->(2) + 26 = 32
%
% The count is taken TOKEN BY TOKEN, so the token that reaches the limit
% is shown whole and may run past it -- `\outputpenalty ' takes the tally
% from 27 to 42. And where the parameter text alone fills the thirty-two,
% the `->' is not shown at all.
%
% A RESTORE DOES NOT START A LINE. Where a \tracingcommands line moves to
% a fresh line first, a restore is written where the last thing left off:
% `[C exactly at the boundary]{restoring \toks3=...' on the one line.
%
% A SAVE TAKEN OVER BY A GLOBAL ASSIGNMENT says `retaining' and names the
% value that stands, not `restoring' with the one saved. Measured in
% tests/trip/probes/when-a-restore-is-retained-instead.tex, which also
% withdraws what this probe used to say about WHEN a save is traced: that
% was an artifact of grepping for `^{', and a {restoring line has no
% newline before it.
\tracingonline=1
\font\ff=cmr10
\def\a{A}\catcode`\[=12 \count5=7 \dimen6=1pt \skip7=2pt plus 1fil
\muskip8=3mu \toks9={x}\hbadness=3 \parindent=4pt
\baselineskip=10pt \thinmuskip=5mu \everypar={p}\textfont2=\ff
\tracingrestores=1
{\def\a{B}\catcode`\[=1 \count5=9 \dimen6=2pt \skip7=3pt \muskip8=4mu
 \toks9={y}\hbadness=4 \parindent=5pt \baselineskip=11pt \thinmuskip=6mu
 \everypar={q}\textfont2=\nullfont \def\new{N}}
\end
