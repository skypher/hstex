% WHAT A MU QUANTITY IS WORTH IN POINTS.
%
% A mu is 1/18 of the quad of the symbol font of the size in hand, so a mu
% glue put into a horizontal list is scaled by that. AN INFINITE PART IS
% NOT: a fil is a fil whatever the quad is, and only a finite stretch or
% shrink is a length the mu stands for. Measured against pdftex with
% cmsy10 as \textfont2, where the quad makes 1mu = 0.55554pt:
%
%   \thinmuskip=1mu plus 2fill minus 3mu
%     -> \glue(\thinmuskip) 0.55554 plus 2.0fill minus 1.66663
%   \medmuskip=1mu plus 2mu minus 3fil
%     -> \glue(\medmuskip) 0.55554 plus 1.11108 minus 3.0fil
%
% and the same of an \mskip the document writes itself.
%
% This engine scaled every part alike, so an infinite stretch came out
% multiplied by the quad. trip line 51 writes
% `\thinmuskip 1mu plus 2fill minus 3mu'.
\catcode`\{=1 \catcode`\}=2 \catcode`\$=3 \catcode`\^=7
\tracingonline=1 \showboxdepth=9 \showboxbreadth=9
\font\r=cmr10 \font\s=cmsy10 \font\x=cmex10 \r
\textfont0=\r \scriptfont0=\r \scriptscriptfont0=\r
\textfont1=\r \scriptfont1=\r \scriptscriptfont1=\r
\textfont2=\s \scriptfont2=\s \scriptscriptfont2=\s
\textfont3=\x \scriptfont3=\x \scriptscriptfont3=\x
\thinmuskip=1mu plus 2fill minus 3mu
\medmuskip=1mu plus 2mu minus 3fil
\message{[A a thin space with an infinite stretch]}
\setbox0=\hbox{$A\mathop{B}C$}
\showbox0
\message{[B a medium space with an infinite shrink]}
\setbox0=\hbox{$A\mathbin{B}C$}
\showbox0
\message{[C a thin space in a script]}
\setbox0=\hbox{$A^{B\mathop{C}D}$}
\showbox0
\message{[D an \mskip written out]}
\setbox0=\hbox{$A\mskip 1mu plus 2fill minus 3mu B$}
\showbox0
\message{[E and one taken from a parameter]}
\setbox0=\hbox{$A\mskip\medmuskip B$}
\showbox0
\message{[done]}
\end
