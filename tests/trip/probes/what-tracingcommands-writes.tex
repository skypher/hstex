% \tracingcommands.
%
% It was the single largest gap in trip's log: 1988 of the oracle's 7292
% lines are these, against none here. This engine now writes 1611 of them
% and the whole log has gone from 4052 lines to 5665.
%
% WHAT IS TRACED, by level:
%   1   every command the main loop obeys -- the unexpandable ones
%   2+  conditionals as well: the \if itself, then {true} or {false}, and
%       the \else and \fi that follow
%
% THE LINE. `{' then, when the mode has changed since the last one traced,
% the mode and a colon and a space, then the command, then `}':
%
%   {vertical mode: \font}
%   {select font cmr10}
%   {\setbox}
%   {restricted horizontal mode: the letter A}
%   {blank space  }
%   {math shift character $}
%   {math mode: the letter x}
%   {superscript character ^}
%   {end-group character }}
%   {internal vertical mode: \shipout}
%
% The command is named the way an error names it -- `the letter A', `blank
% space  ' with its two spaces, `math shift character $', `\relax'. The
% modes are: vertical mode, internal vertical mode, horizontal mode,
% restricted horizontal mode, math mode, display math mode. HSTeX already
% names both: describe_token() for the command and the mode names at
% src/engine.c around 15140.
%
% SELECTING A FONT gets a line of its own, `{select font cmr10}', in place
% of naming the command.
%
% A command that starts a paragraph is traced TWICE, once in the mode it
% was met in and once in horizontal mode, because it is put back and read
% again after \everypar. This engine puts it back the same way, so it
% would fall out.
%
% THREE RULES THAT ARE NOT IN THE FORMAT AND HAD TO BE MEASURED:
%
%   * A RUN OF CHARACTERS IS TRACED ONCE, at its first. `\hbox{ab}' traces
%     the a and not the b; `hello' traces only the h. The reference's main
%     loop stays inside itself for the rest and never comes back to the
%     tracing.
%   * A CHANGE OF MODE STARTS A NEW RUN, which is what traces twice the
%     character that began a paragraph -- once where it was met and once
%     after \everypar.
%   * THE TOKEN THAT FILLS A MATH FIELD IS NOT TRACED, being read by the
%     field machinery rather than obeyed: `$x^2$' traces the x and the ^
%     and not the 2.
%
% And one thing the reference traces that this engine had no token for:
% `{end of alignment template}', written where an entry ends.
%
% LEVEL TWO IS NOT DONE, and here is what it wants and where it snagged.
% At two and above every EXPANDABLE command is traced as well, and a
% conditional is followed by {true} or {false}:
%
%   \message{\number5 \romannumeral7 }
%   \count1=5 \ifnum1<2 \relax\else\relax\fi
%
%   level 1: {\message} {blank space  } {\count} {\relax} {\end}
%   level 2: {\message} {\number} {\romannumeral} {blank space  }
%            {\count} {\ifnum} {true} {\relax} {\else} {\end}
%
% A \write's text is expanded in a mode of its own, written `no mode':
%   {\immediate} {no mode: \number} {vertical mode: blank space  }
%
% TRIED: tracing at the top of the expander's dispatch in
% hstex_engine_next_expanded. It traces the expandable ones correctly and
% ALSO traces every unexpandable one a second time, since they pass
% through the expander on their way to the main loop and are traced again
% there. What is wanted is a test for `this command is one the expander
% will expand', which hstex has nowhere -- the expander is a chain of ifs
% ending in a fallthrough that returns the token. Either enumerate the
% expandable commands in a switch beside the chain, or trace inside each
% branch. Backed out rather than half-written.
%
% \tracingrestores is the companion and is also missing: 257 lines of
% {restoring \x=...} in the oracle. Its format is in
% docs/DECISIONS.md-adjacent probes -- {restoring \a=macro:->A},
% {restoring \count5=7}, {restoring \catcode91=1}, {restoring
% \newone=undefined}.
\tracingonline=1\tracingcommands=1
\font\tt=cmr10 \tt
\setbox0=\hbox{A $x^2$ \relax}
\end
