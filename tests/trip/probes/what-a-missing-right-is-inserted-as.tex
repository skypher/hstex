% WHAT A MISSING \right IS INSERTED AS.
%
% A formula that ends with a \left still open draws `Missing \right.
% inserted', and the context under it reads
%
%   <inserted text>
%                   \right .
%   <to be read again>
%                      $
%
% ONE frame for the two tokens put in, and another for the one that was
% put back. What is PUT IN is an inserted text however many tokens it is;
% what is put BACK is a frame of its own. This engine put all three back
% one at a time and drew three frames.
%
% trip line 298 leaves two \left's open at a $.
\catcode`\{=1 \catcode`\}=2 \catcode`\$=3
\tracingonline=1 \errorcontextlines=5
\font\r=cmr10 \font\s=cmsy10 \font\x=cmex10 \r
\textfont0=\r \scriptfont0=\r \scriptscriptfont0=\r
\textfont1=\r \scriptfont1=\r \scriptscriptfont1=\r
\textfont2=\s \scriptfont2=\s \scriptscriptfont2=\s
\textfont3=\x \scriptfont3=\x \scriptscriptfont3=\x
\message{[A a formula that ends with a \left still open]}
\setbox0=\hbox{$\left(A$}
\message{[B and one closed by a brace]}
\setbox0=\hbox{$\left(A\hbox{}$}
\message{[done]}
\end
