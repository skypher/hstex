% An entry that \span's several columns is set to the FIRST of them, and the
% columns it covers stand behind it as EMPTY boxes of their own widths, each
% behind the \tabskip that would have come before it.
\catcode`\{=1 \catcode`\}=2 \catcode`\#=6 \catcode`\&=4
\tracingonline=1 \showboxbreadth=99 \showboxdepth=99
\tabskip=3pt
\message{[A a row that spans two of three columns]}
\setbox0=\vbox{\halign{\hbox to 10pt{}#&\hbox to 20pt{}#&\hbox to 30pt{}#\cr
  a\span b&c\cr \hbox to 40pt{}&\hbox to 50pt{}&\hbox to 60pt{}\cr}}
\showbox0
\message{[B one that spans all three]}
\setbox0=\vbox{\halign{#&#&#\cr
  \hbox to 5pt{}\span\span\cr \hbox to 40pt{}&\hbox to 50pt{}&\hbox to 60pt{}\cr}}
\showbox0
\message{[C tabskips that differ from column to column]}
\tabskip=1pt
\setbox0=\vbox{\halign{\hbox to 10pt{}#\tabskip=2pt&\hbox to 20pt{}#\tabskip=3pt&\hbox to 30pt{}#\tabskip=4pt\cr
  a\span b&c\cr \hbox to 40pt{}&\hbox to 50pt{}&\hbox to 60pt{}\cr}}
\showbox0
\message{[D one with glue, which fills all the columns it covers]}
\tabskip=3pt
\setbox0=\vbox{\halign{#&#&#\cr
  \hbox to 5pt{}\hskip 0pt plus 1fil\span \hbox to 5pt{}&c\cr
  \hbox to 40pt{}&\hbox to 50pt{}&\hbox to 60pt{}\cr}}
\showbox0
\message{[E the same where the row itself has to shrink]}
\tabskip=3pt minus 2pt
\setbox0=\vbox{\halign to 120pt{#&#&#\cr
  \hbox to 5pt{}\hskip 0pt plus 1fil\span \hbox to 5pt{}&c\cr
  \hbox to 40pt{}&\hbox to 50pt{}&\hbox to 60pt{}\cr}}
\showbox0
\message{[done]}
\end
