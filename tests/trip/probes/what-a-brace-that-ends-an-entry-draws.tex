% A `}' that would close an alignment entry's own group closes nothing the
% document opened: the reference hands the brace back, guesses a \cr was
% meant, and puts one in front of it -- and what follows that \cr is a
% misplaced one, since the entry it would have ended is already gone.
\catcode`\{=1 \catcode`\}=2 \catcode`\#=6 \catcode`\&=4
\tracingonline=1 \errorcontextlines=5
\message{[A a brace where the entry wanted a \string\cr]}
\setbox0=\vbox{\halign{#\cr \tracingcommands=1 }\cr\tracingcommands=0 }}
\message{[done]}
\end
