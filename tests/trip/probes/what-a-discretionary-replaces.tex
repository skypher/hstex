% What a discretionary replaces sits on the enclosing list behind it, and
% belongs to it -- \unkern and its relatives leave it alone. What may not
% stand in a discretionary part at all is deleted, and shown as it goes.
\catcode`\{=1 \catcode`\}=2
\font\tenrm=cmr10 \tenrm
\tracingonline=1 \hsize=100pt \parindent=0pt
\message{[A a kern inside a discretionary's replace text]}
\setbox0=\hbox{A\discretionary{}{}{B\kern2pt}\unkern\showthe\lastkern}
\message{[B a kern on the list itself]}
\setbox0=\hbox{A\kern2pt\unkern\showthe\lastkern}
\message{[C a kern after the replace text]}
\setbox0=\hbox{A\discretionary{}{}{B}\kern2pt\unkern\showthe\lastkern}
\message{[D glue inside the replace text]}
\setbox0=\hbox{A\discretionary{}{}{B\hskip2pt}\unskip\showthe\lastskip}
\message{[E a penalty inside the replace text]}
\setbox0=\hbox{A\discretionary{}{}{B\penalty7}\unpenalty\showthe\lastpenalty}
\message{[F three nodes after the offender]}
\setbox0=\hbox{A\discretionary{}{}{B\hskip2pt C\kern3pt\lower1pt\hbox{D}}}
\showboxbreadth=100 \showboxdepth=100
\message{[G a nested box after the offender]}
\setbox0=\hbox{\discretionary{X\penalty5 \hbox{Y}}{}{}}
\showboxbreadth=-1 \showboxdepth=-1
\message{[H nothing may be shown of it at all]}
\setbox0=\hbox{A\discretionary{}{}{B\hskip2pt C\hbox{D}}}
\showboxbreadth=2 \showboxdepth=0
\message{[I past the breadth]}
\setbox0=\hbox{\discretionary{\penalty1 \penalty2 \penalty3 \penalty4 }{}{}}
\message{[done]}
\end
