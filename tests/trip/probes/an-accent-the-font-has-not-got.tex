% AN ACCENT THE FONT HAS NOT GOT IS DROPPED, AND SO IS ITS WAIT.
%
% \accent reads a character number and then goes on waiting for the
% character it will go over, reading blanks, \relax and assignments in
% the meantime. None of that happens when the font has no such accent:
% the accent is dropped, nothing waits, the space factor is left where it
% stood, and whatever follows is read by the main loop and traced there.
%
%   \hbox{f\accent200 i}        the fi ligature does NOT form
%   \hbox{A\accent200 ...}      \the\spacefactor is 999, not 1000
%   \hbox{\accent200 A}         {the letter A} is traced
%
% The ligature run is broken all the same, but only because \accent is a
% command and the word ended when it was met.
%
% HSTeX left the accent waiting, so the token after it was swallowed
% rather than obeyed. trip line 322 writes `\accent\x' with \chardef\x200
% in a font that has no 200, and the blank that the \space after it
% expands to went missing.
\catcode`\{=1 \catcode`\}=2
\font\ff=cmr10 \ff
\tracingonline=1 \showboxbreadth=99 \showboxdepth=99
\hsize=100pt \hbadness=10000 \hfuzz=100pt
\chardef\x200
\def\sp{ }
\message{[A a missing accent does not swallow what follows]}
\tracingcommands=2
\setbox0=\hbox{\accent\x\sp\accent`A B}
\tracingcommands=0
\message{[B an accent the font does have, for contrast]}
\tracingcommands=2
\setbox0=\hbox{\accent`A\sp\accent`A B}
\tracingcommands=0
\message{[C the character after a missing accent is read plainly]}
\tracingcommands=2
\setbox0=\hbox{\accent\x A}
\tracingcommands=0
\message{[D a missing accent breaks the fi ligature]}
\setbox0=\hbox{f\accent200 i}\showbox0
\message{[E and leaves the space factor alone]}
\sfcode`A=999
\setbox0=\hbox{A\accent200 \message{[<\the\spacefactor>]}}
\message{[F where one the font has sets it to 1000]}
\setbox0=\hbox{A\accent`B C\message{[<\the\spacefactor>]}}
\message{[G the fi ligature on its own]}
\setbox0=\hbox{fi}\showbox0
\message{[done]}
\end
