% A LARGE OPERATOR IS SET IN A BOX WHATEVER BECAME OF ITS NUCLEUS.
%
% \mathop puts its nucleus in a box of its own and hangs that box on the
% axis. Where the font has not got the character the nucleus names, there is
% nothing to put in the box -- and the reference makes the box anyway, empty,
% and hangs it on the axis just the same:
%
%   \hbox(0.0+0.0)x0.0, shifted -2.5
%
% where 2.5pt is \fontdimen22 of \textfont2. HSTeX drew nothing at all, which
% is what trip's line 275 caught: it writes `\mathop y' in a font with no y,
% and the empty box that comes of it was the one box the reference drew in
% trip's second pass that HSTeX did not.
%
% ONLY \mathop. Every other class draws nothing when the character is
% missing, which is what [A] measures: ord, bin, rel, open, close, punct and
% inner all come to nothing, and op alone comes to a box. That is the whole
% of the rule, and it is why the box cannot be found by looking at what an
% empty box shifted to the axis usually is -- a delimiter the fonts could not
% supply. Those are \nulldelimiterspace wide; see
% what-a-delimiter-that-is-not-there-comes-out-as.tex, where nineteen of them
% agree and this one was ruled out by being the only one of no width at all.
%
% [B] is trip line 275's own shape, `\mathop y' with the \nonscript pair
% behind it, and [C] asks the same of an operator that carries limits and of
% one that carries scripts, since a box that is not there cannot carry
% either.
\catcode`\{=1 \catcode`\}=2 \catcode`\$=3 \catcode`\^=7 \catcode`\_=8
\tracingonline=1 \showboxbreadth=99 \showboxdepth=99
\font\tenrm=cmr10 \font\tensy=cmsy10 \font\tenex=cmex10
\textfont0=\tenrm \scriptfont0=\tenrm \scriptscriptfont0=\tenrm
\textfont1=\nullfont \scriptfont1=\nullfont \scriptscriptfont1=\nullfont
\textfont2=\tensy \scriptfont2=\tensy \scriptscriptfont2=\tensy
\textfont3=\tenex \scriptfont3=\tenex \scriptscriptfont3=\tenex
\tenrm \tracinglostchars=1
\message{[A every class in turn, with a character the font has not got]}
\setbox0=\hbox{$\mathord y$}\showbox0
\setbox0=\hbox{$\mathop y$}\showbox0
\setbox0=\hbox{$\mathbin y$}\showbox0
\setbox0=\hbox{$\mathrel y$}\showbox0
\setbox0=\hbox{$\mathopen y$}\showbox0
\setbox0=\hbox{$\mathclose y$}\showbox0
\setbox0=\hbox{$\mathpunct y$}\showbox0
\setbox0=\hbox{$\mathinner y$}\showbox0
\message{[B trip line 275's own shape]}
\setbox0=\hbox{$\mathop y\nonscript\textstyle\nonscript\mskip9mu minus1fil x$}
\showbox0
\message{[C the same carrying limits, and in display style where it stacks]}
\setbox0=\hbox{$\mathop y\limits$}\showbox0
\setbox0=\hbox{$\mathop y\nolimits$}\showbox0
\setbox0=\hbox{$\displaystyle\mathop y$}\showbox0
\message{[D and an empty \mathop, which has no character to be missing]}
\setbox0=\hbox{$\mathop{}$}\showbox0
\message{[done]}
\end
