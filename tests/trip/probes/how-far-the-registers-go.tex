\message{[D a negative register]}
\count-1=1
\message{[E dimen past the end]}
\dimen300=1pt
\message{[F skip past the end]}
\skip256=1pt
\message{[G muskip past the end]}
\muskip256=1mu
\message{[H toks past the end]}
\toks256={x}
\message{[I box past the end]}
\setbox256=\hbox{}

\showthe\count0 \showthe\tk
